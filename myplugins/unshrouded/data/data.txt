mission "Untangle the shroud"
	job
	repeat
	description `Untangle the shroud?`
	to offer
		not "event: Untangle"
	on accept
		conversation
			`Untangled Shroud.`
			action
				event "Untangle"
				clear "event: Tangle"
				fail
